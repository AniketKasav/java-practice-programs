-------------------------------------------------------------------------------------------------------------

Description

Each employee has a salary value. When you apply the condition “greater than 30,000 AND less than 70,000,” the system checks all records and keeps only those employees whose salary falls between these two values.

This means:

Salaries above 30,000 are included

Salaries below 70,000 are included

Salaries 30,000 or less and 70,000 or more are excluded

Important: BETWEEN includes both values
BETWEEN 30000 AND 70000

means:

salary >= 30000
AND
salary <= 70000

If you want salary strictly greater than 30000 AND strictly less than 70000:

SELECT * FROM employees
WHERE salary > 30000 AND salary < 70000;

Salary > 40,000 must always be true (because AND has higher precedence than OR if no brackets are specified)

********************************************************************************************
1. Customer Table
CREATE TABLE customer (
customer_id INT AUTO_INCREMENT PRIMARY KEY,
name VARCHAR(100) NOT NULL,
email VARCHAR(150) UNIQUE NOT NULL,
phone VARCHAR(15) UNIQUE,
address TEXT NOT NULL,
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

2. Category Table
CREATE TABLE category (
category_id INT AUTO_INCREMENT PRIMARY KEY,
category_name VARCHAR(100) NOT NULL UNIQUE
);

3. Product Table
CREATE TABLE product (
product_id INT AUTO_INCREMENT PRIMARY KEY,
product_name VARCHAR(200) NOT NULL,
price DECIMAL(10,2) NOT NULL CHECK (price > 0),
stock INT NOT NULL CHECK (stock >= 0),
category_id INT,
FOREIGN KEY (category_id) REFERENCES category(category_id)
ON DELETE SET NULL
ON UPDATE CASCADE
);

4. Orders Table
CREATE TABLE orders (
order_id INT AUTO_INCREMENT PRIMARY KEY,
customer_id INT NOT NULL,
order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
status VARCHAR(50) DEFAULT 'Pending',
FOREIGN KEY (customer_id) REFERENCES customer(customer_id)
ON DELETE CASCADE
ON UPDATE CASCADE
);

5. Order_Items Table
CREATE TABLE order_items (
order_id INT,
product_id INT,
quantity INT NOT NULL CHECK (quantity > 0),
price DECIMAL(10,2) NOT NULL CHECK (price > 0),
PRIMARY KEY (order_id, product_id),
FOREIGN KEY (order_id) REFERENCES orders(order_id)
ON DELETE CASCADE,
FOREIGN KEY (product_id) REFERENCES product(product_id)
ON DELETE CASCADE
);

6. Payment Table
CREATE TABLE payment (
payment_id INT AUTO_INCREMENT PRIMARY KEY,
order_id INT UNIQUE,
payment_method VARCHAR(50) NOT NULL,
payment_status VARCHAR(50) DEFAULT 'Pending',
amount DECIMAL(10,2) NOT NULL CHECK (amount > 0),
FOREIGN KEY (order_id) REFERENCES orders(order_id)
ON DELETE CASCADE
);

7. Shipping Table
CREATE TABLE shipping (
shipping_id INT AUTO_INCREMENT PRIMARY KEY,
order_id INT UNIQUE,
address TEXT NOT NULL,
city VARCHAR(100) NOT NULL,
postal_code VARCHAR(10) NOT NULL,
delivery_status VARCHAR(50) DEFAULT 'Processing',
FOREIGN KEY (order_id) REFERENCES orders(order_id)
ON DELETE CASCADE
);

8.Review Table
CREATE TABLE review (
review_id INT AUTO_INCREMENT PRIMARY KEY,
customer_id INT,
product_id INT,
rating INT CHECK (rating BETWEEN 1 AND 5),
comment TEXT,
review_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
FOREIGN KEY (customer_id) REFERENCES customer(customer_id)
ON DELETE CASCADE,
FOREIGN KEY (product_id) REFERENCES product(product_id)
ON DELETE CASCADE
);

    ->     c.customer_id,
